def send_to_syslog(alerts, syslog_host, syslog_port):
    for alert in alerts['data']:
        syslog_message = f"<6>1 {alert['timestamp']} {socket.gethostname()} AdaptiveShield: {alert['description']}"
        send_syslog(syslog_message, syslog_host, syslog_port)
        print(f"Event sent to syslog: {syslog_message}")
        time.sleep(1)  # Sleep for 1 second between events to avoid flooding the syslog server

def send_syslog(message, host, port):
    syslog_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    syslog_socket.sendto(message.encode('utf-8'), (host, port))
    syslog_socket.close()
