

import os
import glob
import socket
from pathlib import Path
import heapq

def send_logs(log_files, server_ip, server_port):
    with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock:
        for log_file in log_files:
            with open(log_file, 'r') as file:
                log_data = file.read()
                sock.sendto(log_data.encode('utf-8'), (server_ip, server_port))

def get_newest_logs(directory, num_logs):
    log_files = glob.glob(os.path.join(directory, '*.log'))
    newest_logs = heapq.nlargest(num_logs, log_files, key=os.path.getctime)
    return newest_logs

if __name__ == "__main__":
    log_directory = 'xxxxx'
    syslog_server_ip = 'xxxxx'
    syslog_server_port = xxxxx
    num_logs_to_send = 10

    newest_logs = get_newest_logs(log_directory, num_logs_to_send)

    if newest_logs:
        send_logs(newest_logs, syslog_server_ip, syslog_server_port)
        print(f"Sent the {num_logs_to_send} newest logs to {syslog_server_ip}:{syslog_server_port}")
    else:
        print("No log files found in the specified directory.")
